package com.ubs.efm.docuflow.control.config;

import com.flowable.control.spring.security.web.ControlSecurityCustomizer;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.Customizer;

@Configuration
public class ControlSecurityConfiguration {

    @Bean
    public ControlSecurityCustomizer controlSecurityCustomizer() {
        return http -> http
                .httpBasic(Customizer.withDefaults())
                .formLogin(Customizer.withDefaults());
    }
}
